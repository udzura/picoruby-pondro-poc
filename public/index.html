Plain PONDRO

class Counter < Pondro::Object
  state :count, default: 0
  rpc :increment, :value, http: true
end

PONDRO + WebSocket

class ChatRoom < Pondro::Object
  use Pondro::WebSocket
  def on_message(socket, message)
    pending = Counter[counter_id].increment
    count = pending.await
  end
end
